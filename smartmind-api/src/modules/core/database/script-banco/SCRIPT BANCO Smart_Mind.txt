CREATE TABLE "Consumo" (
	id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
	consumo DECIMAL(10,3),
	dt_medicao TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
	fk_dispositivo INTEGER NOT NULL,

	CONSTRAINT fk_consumo_dispositivo
		FOREIGN KEY (fk_dispositivo)
		REFERENCES "Dispositivo" (id)
);

CREATE INDEX IF NOT EXISTS idx_consumo_dispositivo_data ON "Consumo" (fk_dispositivo, dt_medicao);

INSERT INTO "Consumo" (consumo, dt_medicao, fk_dispositivo)
SELECT ROUND(((d.potencia_watts / 1000.0) * (d.horas_uso_dia / 24.0) * (0.6 + random() * 0.8))::numeric, 3),
       h,
       d.id
FROM "Dispositivo" d
CROSS JOIN generate_series(
    date_trunc('hour', LOCALTIMESTAMP) - INTERVAL '120 days',
    date_trunc('hour', LOCALTIMESTAMP),
    INTERVAL '1 hour'
) AS h
WHERE d.ativo = TRUE
  AND d.potencia_watts > 0;

CREATE TABLE "Chat_Mensagem" (
	id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
	mensagem TEXT NOT NULL,
	resposta TEXT NOT NULL,
	intencao VARCHAR(100),
	dt_mensagem TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
	fk_usuario INTEGER NOT NULL,
	fk_casa INTEGER NOT NULL,
	fk_conversa INTEGER NULL,
	dados JSONB NULL,

	CONSTRAINT fk_chatMensagem_usuario
		FOREIGN KEY (fk_usuario)
		REFERENCES "Usuario" (id),

	CONSTRAINT fk_chatMensagem_casa
		FOREIGN KEY (fk_casa)
		REFERENCES "Casa" (id),

	CONSTRAINT fk_chatMensagem_conversa
    	FOREIGN KEY (fk_conversa)
    	REFERENCES "Chat_Conversa" (id)
);
